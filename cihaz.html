ESP32 firmware ve sistem kodu
// EcoAcoustic-AI Ana Firmware #include "edge_impulse_model.h" void setup() { Serial.begin(115200); setupLoRa(); setupI2S(); run_classifier_init(); } void loop() { captureAudio(); float conf; int threat = runInference(audio_buffer, &conf); if (threat > 0 && conf > 0.85) { sendAlert(threat, conf); } delay(100); }
// LoRa 868 MHz iletişim #define LORA_SS 18 #define LORA_RST 14 #define LORA_DIO0 26 void setupLoRa() { LoRa.setPins(LORA_SS, LORA_RST, LORA_DIO0); LoRa.begin(868E6); LoRa.setSpreadingFactor(7); LoRa.setSignalBandwidth(125E3); LoRa.setTxPower(20); } void sendAlert(int type, float conf) { LoRa.beginPacket(); LoRa.print("ECOAI|NODE=ECO-04|"); LoRa.print("TYPE="); LoRa.print(type); LoRa.print("|CONF="); LoRa.print(conf); LoRa.endPacket(); }
// TinyML çıkarım (Edge Impulse) int runInference(float* audio, float* conf) { signal_t signal; signal.total_length = SAMPLE_BUFFER; signal.get_data = [](size_t offset, size_t length, float* out) { memcpy(out, &audio[offset], length * sizeof(float)); return 0; }; ei_impulse_result_t result; run_classifier(&signal, &result, false); int best = 0; *conf = 0; for (int i = 0; i < EI_CLASSIFIER_LABEL_COUNT; i++) { if (result.classification[i].value > *conf) { *conf = result.classification[i].value; best = i; } } return best; }
// MPU6050 — Sabotaj algılama #include <MPU6050.h> MPU6050 mpu; bool checkTamper() { int16_t ax, ay, az; mpu.getAcceleration(&ax, &ay, &az); float accel = sqrt(ax*ax + ay*ay + az*az) / 16384.0; if (accel > 2.5) return true; if (accel < 0.3) return true; return false; }
